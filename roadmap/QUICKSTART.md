```python
print(f"GPU 사용 가능: {torch.cuda.is_available()}")

    print(f"GPU 이름: {torch.cuda.get_device_name(0)}")
    print("✅ 준비 완료! 학습 가능!")
else:
    print("⚠️ GPU 없음. 런타임 > 런타임 유형 변경 > GPU 선택")
```
